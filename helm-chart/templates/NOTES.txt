1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
  https://{{ index .Values.ingress.hosts 0 "host" }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "go-sse.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "go-sse.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "go-sse.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "go-sse.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. Check the deployment status:
  kubectl rollout status deployment/{{ include "go-sse.fullname" . }} -n {{ .Release.Namespace }}

3. To view pod logs:
  kubectl logs -f deployment/{{ include "go-sse.fullname" . }} -n {{ .Release.Namespace }}

4. Health checks used by the probes:
  /livez  (liveness and startup)   /readyz (readiness)   /health (human-friendly status)

5. Observability:
  - Prometheus metrics: /metrics{{ if .Values.serviceMonitor.enabled }} (scraped by the ServiceMonitor){{ end }}
  - JSON logs on stdout, level {{ .Values.logLevel }}
  {{- if and .Values.otel.enabled .Values.otel.endpoint }}
  - Traces and logs exported via OTLP to {{ .Values.otel.endpoint }}
  {{- else }}
  - OTLP export of traces/logs disabled (set otel.enabled=true and otel.endpoint)
  {{- end }}
  {{- if .Values.grafanaDashboard.enabled }}
  - Grafana dashboard ConfigMap: {{ include "go-sse.fullname" . }}-dashboard
  {{- end }}
